--[[
RAVENA HUB | Murder Mystery-style game (YOUR OWN Roblox experience)
Install as a LocalScript in StarterPlayer > StarterPlayerScripts.

Included:
- Black/purple responsive-style UI with minimize/reopen
- Opens directly to the main panel; no loading screen
- Team ESP: Sheriff/Sharif/Xerife blue, Murderer/Assassino red, Innocent/Inocente green
- Speed control (client-side prototype; production games should validate server-side)
- Noclip toggle (client-side prototype; production games should authorize server-side)
- Auto Farm Coins: looks in Workspace.Coins and walks toward one at a time;
  stops pursuing a coin if it disappears, then searches for another.
- Shoot Murder button: shown only when the local player's Team is Sheriff-like.
  It requests a shot through ReplicatedStorage.RavenaShootMurder (RemoteEvent).
  You must implement and validate this RemoteEvent on the SERVER for your own game.
- Credits and Discord open button.

Important:
A LocalScript cannot securely award coins or validate weapon damage. The coin collection
must be handled by your game's existing server-side touch/collection system. Clipboard
access is not available to normal Roblox LocalScripts; Discord button opens the invite.
]]

-- ESP
local espEnabled = false
local espObjects = {}
local function removeEsp(player)
    local h = espObjects[player]
    if h then h:Destroy() end
    espObjects[player] = nil
end
local function updateEsp(player)
    if player == LocalPlayer then return end
    local character = player.Character
    if not espEnabled or not character then
        removeEsp(player)
        return
    end
    local color = roleColor(player)
    if not color then
        removeEsp(player)
        return
    end
    local h = espObjects[player]
    if not h or h.Parent ~= character then
        removeEsp(player)
        h = Instance.new("Highlight")
        h.Name = "RavenaTeamESP"
        h.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        h.Parent = character
        espObjects[player] = h
    end
    h.FillColor = color
    h.OutlineColor = color
    h.FillTransparency = 0.72
    h.OutlineTransparency = 0.05
end

local function refreshEsp()
    for _, p in ipairs(Players:GetPlayers()) do updateEsp(p) end
end

Players.PlayerAdded:Connect(function(p)
    p.CharacterAdded:Connect(function()
        task.wait(0.4)
        updateEsp(p)
    end)
    p:GetPropertyChangedSignal("Team"):Connect(function() updateEsp(p) end)
end)
Players.PlayerRemoving:Connect(removeEsp)
for _, p in ipairs(Players:GetPlayers()) do
    p.CharacterAdded:Connect(function()
        task.wait(0.4)
        updateEsp(p)
    end)
    p:GetPropertyChangedSignal("Team"):Connect(function() updateEsp(p) end)
end

-- Movement controls
local noclipEnabled = false
local speedEnabled = false
local speedValue = 24
local originalCollisions = {}
local noclipConnection

local function getCharacterParts()
    local character = LocalPlayer.Character
    if not character then return {} end
    local parts = {}
    for _, obj in ipairs(character:GetDescendants()) do
        if obj:IsA("BasePart") then table.insert(parts, obj) end
    end
    return parts
end

local function setNoclip(enabled)
    noclipEnabled = enabled
    if noclipConnection then noclipConnection:Disconnect(); noclipConnection = nil end
    if enabled then
        noclipConnection = RunService.Stepped:Connect(function()
            for _, part in ipairs(getCharacterParts()) do
                if part.Name ~= "HumanoidRootPart" and originalCollisions[part] == nil then
                    originalCollisions[part] = part.CanCollide
                end
                part.CanCollide = false
            end
        end)
    else
        for part, canCollide in pairs(originalCollisions) do
            if part and part.Parent then part.CanCollide = canCollide end
        end
        table.clear(originalCollisions)
    end
end

local function applySpeed()
    local character = LocalPlayer.Character
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    if humanoid then humanoid.WalkSpeed = speedEnabled and speedValue or 16 end
end

LocalPlayer.CharacterAdded:Connect(function()
    task.wait(0.5)
    if speedEnabled then applySpeed() end
    if noclipEnabled then setNoclip(true) end
end)

-- Auto farm coins. Relies on your own game's existing coin touch/collection system.
local autoFarm = false
local farmToken = 0
local function findCoinsFolder()
    return workspace:FindFirstChild(COINS_FOLDER_NAME)
end
local function coinPart(obj)
    if obj:IsA("BasePart") then return obj end
    if obj:IsA("Model") then
        return obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart", true)
    end
    return nil
end
local function getRootAndHumanoid()
    local char = LocalPlayer.Character
    if not char then return nil, nil end
    return char:FindFirstChild("HumanoidRootPart"), char:FindFirstChildOfClass("Humanoid")
end
local function runFarm(token)
    while autoFarm and token == farmToken do
        local folder = findCoinsFolder()
        local target
        if folder then
            for _, coin in ipairs(folder:GetChildren()) do
                local part = coinPart(coin)
                if part and part:IsDescendantOf(workspace) then
                    target = part
                    break
                end
            end
        end
        if target then
            local root, humanoid = getRootAndHumanoid()
            if root and humanoid and humanoid.Health > 0 then
                humanoid:MoveTo(target.Position)
                local start = os.clock()
                while autoFarm and token == farmToken and target.Parent and os.clock() - start < 5 do
                    root, humanoid = getRootAndHumanoid()
                    if not root or not humanoid then break end
                    if (root.Position - target.Position).Magnitude <= 4 then break end
                    task.wait(0.15)
                end
            else
                task.wait(0.5)
            end
        else
            task.wait(0.5)
        end
        task.wait(0.15)
    end
end

addPage("Combate", function()
    section("Shoot Murder")
    info("Disponível apenas quando sua equipe é Sheriff/Xerife. O servidor do seu próprio jogo deve validar o disparo.")
    local b = makeButton(content, "Shoot Murder", 42)
    local function sheriffTeam()
        local n = LocalPlayer.Team and normalize(LocalPlayer.Team.Name) or ""
        return n == "sheriff" or n == "sharif" or n == "xerife"
    end
    local function update()
        b.Active = sheriffTeam()
        b.AutoButtonColor = sheriffTeam()
        b.Text = sheriffTeam() and "Shoot Murder — solicitar disparo" or "Disponível apenas para Sheriff"
        b.BackgroundColor3 = sheriffTeam() and COLORS.purple2 or COLORS.panel2
    end
    update()
    LocalPlayer:GetPropertyChangedSignal("Team"):Connect(update)
    b.MouseButton1Click:Connect(function()
        if not sheriffTeam() then return end
        local remote = ReplicatedStorage:FindFirstChild("RavenaShootMurder")
        if remote and remote:IsA("RemoteEvent") then
            remote:FireServer()
        else
            b.Text = "Configure RavenaShootMurder no servidor"
            task.delay(2.5, update)
        end
    end)
end)

-- Drag window from top bar
do
    local dragging, dragStart, startPos
    top.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = main.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end
